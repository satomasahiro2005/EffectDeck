cmake_minimum_required(VERSION 3.24)
project(effectdeck_host_tests LANGUAGES C)
enable_testing()
add_executable(preview_tone preview_tone.c ../../Sources/Shared/ETPreviewTone.c)
set_property(TARGET preview_tone PROPERTY C_STANDARD 11)
target_include_directories(preview_tone PRIVATE ../../Sources/Shared)
if(MSVC)
  target_compile_options(preview_tone PRIVATE /experimental:c11atomics /utf-8 /UNDEBUG)
else()
  target_compile_options(preview_tone PRIVATE -UNDEBUG)
  target_link_libraries(preview_tone PRIVATE m)
endif()
add_test(NAME preview_tone COMMAND preview_tone)
